import { Chat } from '/snippets/chat.jsx';

يستطيع الذكاء الاصطناعي لديك أن يرسل إليك كل ما يتسع له WhatsApp.

## كيف يبدو عمليًا

<Chat
  messages={[
    { you: true, text: 'أرسل لي لقطة شاشة لنهاية البناء مع الوصف «اكتمل البناء».', time: '16:40' },
    { name: 'Claude Code', media: { kind: 'image' }, text: 'اكتمل البناء', time: '16:41' },
    { you: true, reaction: '👍', text: 'رائع، شكرًا.', time: '16:41' },
  ]}
/>

وتصل الأنواع الأخرى بالطريقة نفسها.

<Chat
  messages={[
    { name: 'Claude Code', media: { kind: 'video', label: 'تسجيل العرض التوضيحي' }, time: '16:42' },
    { name: 'Claude Code', media: { kind: 'voice', label: '0:24' }, time: '16:42' },
    { name: 'Claude Code', media: { kind: 'document', label: 'release-notes.pdf' }, text: 'ملاحظات الإصدار', time: '16:43' },
  ]}
/>

<Chat
  messages={[
    { name: 'Claude Code', media: { kind: 'sticker' }, time: '16:43' },
    { name: 'Claude Code', media: { kind: 'location', label: 'Café Brasileira, Chiado' }, time: '16:44' },
    { name: 'Claude Code', media: { kind: 'contact', label: 'مارتا سوزا', action: 'راسل' }, time: '16:44' },
  ]}
/>

## ما الذي يستطيع إرساله

| ما تراه على هاتفك                       | النوع      |
| --------------------------------------- | ---------- |
| نص، مع تنسيق WhatsApp والرموز التعبيرية | `text`     |
| صورة                                    | `image`    |
| فيديو                                   | `video`    |
| ملف صوتي أو رسالة صوتية                 | `audio`    |
| ملف، مثل PDF                            | `document` |
| ملصق                                    | `sticker`  |
| نقطة على الخريطة                        | `location` |
| بطاقة جهة اتصال أو أكثر                 | `contacts` |
| تفاعل مع إحدى رسائلك                    | `reaction` |
| أزرار أو قائمة تختار منها               | `choice`   |

يمكن لأي منها أن يقتبس إحدى رسائلك. وعندما ترسل صورة أو ملفًا إلى الذكاء الاصطناعي، يتلقاه هو أيضًا.
