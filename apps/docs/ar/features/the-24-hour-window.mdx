import { Chat } from '/snippets/chat.jsx';

لا يسمح WhatsApp لـ zapbip بمراسلتك بحرية إلا خلال 24 ساعة من آخر رسالة أرسلتها إليه.

## النافذة مفتوحة

تصل رسائل الذكاء الاصطناعي لديك فورًا.

<Chat
  messages={[
    { you: true, text: 'أخبرني عندما تنتهي مهمة الليل.', time: '22:10' },
    { name: 'Claude Code', text: 'انتهت مهمة الليل ✅', time: '23:31' },
  ]}
/>

## النافذة مغلقة

تصلك رسالة تنبيه تذكر اسم الذكاء الاصطناعي ومعاينة قصيرة لرسالته. اكتب أي شيء فتصل الرسائل بالترتيب. والرسائل التي مضى عليها أكثر من 7 أيام تنتهي صلاحيتها.

<Chat
  messages={[
    { you: true, text: 'أخبرني عندما تنتهي مهمة الليل.', time: 'أمس 22:10' },
    { notice: true, text: 'لدى الوكيل claude-code رسالة لك: انتهت مهمة الليل ✅', time: '23:31' },
    { you: true, text: 'ok', time: '08:05' },
    { name: 'Claude Code', text: 'انتهت مهمة الليل ✅', time: '08:05' },
  ]}
/>
