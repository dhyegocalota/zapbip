import { Chat } from '/snippets/chat.jsx';

## ما تقوله للذكاء الاصطناعي

<Chat messages={[{ you: true, text: 'قبل أن تحجز، اسألني أي فندق أفضّل وانتظر جوابي.', time: '09:14' }]} />

## ما تراه على هاتفك

تظهر حتى 3 خيارات أزرارًا.

<Chat
  messages={[
    { name: 'Claude', text: 'أي فندق في لشبونة؟', options: ['ألفاما', 'بايشا', 'بيليم'], time: '09:15' },
    { you: true, text: 'بايشا', time: '09:16' },
    { name: 'Claude', text: 'جارٍ حجز الفندق في بايشا لـ 3 ليالٍ.', time: '09:16' },
  ]}
/>

ومن 4 إلى 10 خيارات تظهر قائمة.

<Chat
  messages={[
    {
      name: 'Claude',
      text: 'في أي حي أبحث؟',
      list: { title: 'اختر واحدًا', rows: ['ألفاما', 'بايشا', 'بيليم', 'شيادو', 'برنسيبي ريال'] },
      time: '09:17',
    },
    { you: true, text: 'شيادو', time: '09:17' },
  ]}
/>

اختر أحدها، أو اكتب جوابك مباشرةً.

## إذا تأخرت في الرد

ينتظر الذكاء الاصطناعي لديك ما دام ذلك لازمًا. ولا يضيع جوابك حتى لو جاء متأخرًا.
