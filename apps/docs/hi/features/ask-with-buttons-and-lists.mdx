import { Chat } from '/snippets/chat.jsx';

## आप अपने AI से क्या कहते हैं

<Chat messages={[{ you: true, text: 'बुकिंग से पहले मुझसे पूछना कि कौन-सा होटल पसंद है, और मेरे जवाब का इंतज़ार करना।', time: '09:14' }]} />

## आपको फ़ोन पर क्या दिखता है

3 तक विकल्प बटन के रूप में दिखते हैं।

<Chat
  messages={[
    { name: 'Claude', text: 'लिस्बन के लिए कौन-सा होटल?', options: ['Alfama', 'Baixa', 'Belém'], time: '09:15' },
    { you: true, text: 'Baixa', time: '09:16' },
    { name: 'Claude', text: 'Baixa में 3 रातों की बुकिंग हो रही है।', time: '09:16' },
  ]}
/>

4 से 10 विकल्प लिस्ट के रूप में दिखते हैं।

<Chat
  messages={[
    {
      name: 'Claude',
      text: 'किस इलाके में खोजूँ?',
      list: { title: 'एक चुनें', rows: ['Alfama', 'Baixa', 'Belém', 'Chiado', 'Príncipe Real'] },
      time: '09:17',
    },
    { you: true, text: 'Chiado', time: '09:17' },
  ]}
/>

कोई एक विकल्प चुनें, या बस अपना जवाब लिख दें।

## अगर आप देर से जवाब दें

आपका AI कुछ देर इंतज़ार करता है और इंतज़ार करता रहता है। आपका जवाब कभी नहीं खोता, चाहे वह बाद में पहुँचे।
